# biomod — BIOM Format I/O for Microbiome Data

The BIOM (Biological Observation Matrix) format is the standard file format for microbiome and metagenomics count tables. It stores sample-observation (OTU/ASV/taxon) count matrices with associated metadata in a JSON or HDF5-based format. This skill covers reading, writing, converting, and analyzing BIOM files in Python.

## When to Use

- Reading OTU/ASV tables generated by QIIME 2, mothur, or DADA2
- Converting between BIOM format and pandas DataFrames for downstream analysis
- Writing custom analysis results back to BIOM format for QIIME 2 compatibility
- Accessing sample and feature metadata embedded in BIOM files
- Building microbiome data pipelines that interface with QIIME 2 artifacts

## Quick Start

```python
import biom
import pandas as pd

# Load a BIOM file (JSON or HDF5 format)
table = biom.load_table("otu_table.biom")

# Convert to pandas DataFrame (features x samples)
df = table.to_dataframe(dense=True)
print(f"Table shape: {df.shape[0]} features x {df.shape[1]} samples")

# Get metadata
sample_ids = table.ids(axis="sample")
feature_ids = table.ids(axis="observation")
print(f"Samples: {len(sample_ids)}, Features: {len(feature_ids)}")

# Access per-sample and per-feature metadata
for sid in sample_ids[:3]:
    md = table.metadata(sid, axis="sample")
    print(f"  {sid}: {md}")
```

## Core Capabilities

### 1. Reading and Exploring BIOM Files

```python
import biom

# Load from file
table = biom.load_table("feature_table.biom")

# Basic properties
print(f"Table type: {table.type}")  # "OTU table", "Taxon table", "Function table", etc.
print(f"Density: {table.get_table_density():.4f}")
print(f"Non-zero entries: {table.nnz}")
print(f"Total counts: {table.sum(axis='whole'):.0f}")

# Convert to sparse or dense DataFrame
df_dense = table.to_dataframe(dense=True)
df_sparse = table.to_dataframe(dense=False)
print(f"Dense shape: {df_dense.shape}, Sparse entries: {df_sparse.nnz}")

# Get observation (feature) metadata
obs_md = table.metadata(axis="observation")
if obs_md is not None:
    import pandas as pd
    taxonomy_df = pd.DataFrame([dict(m) for m in obs_md], index=table.ids(axis="observation"))
    print("\nTaxonomy summary:")
    print(taxonomy_df.head())

# Get sample metadata
samp_md = table.metadata(axis="sample")
if samp_md is not None:
    samp_df = pd.DataFrame([dict(m) for m in samp_md], index=table.ids(axis="sample"))
    print("\nSample metadata columns:", list(samp_df.columns))
```

### 2. Creating and Writing BIOM Files

```python
import biom
import numpy as np
import pandas as pd
from scipy.sparse import csr_matrix

# Method 1: From pandas DataFrame
df = pd.DataFrame(
    np.random.poisson(10, (100, 20)),
    index=[f"ASV_{i}" for i in range(100)],
    columns=[f"Sample_{j}" for j in range(20)],
)
table = biom.Table(df.values.T, observation_ids=df.index.tolist(),
                    sample_ids=df.columns.tolist())
biom.util.hdf5.overwrite("output_table.biom", table)

# Method 2: From sparse matrix (memory-efficient for large tables)
sparse_data = csr_matrix(np.random.poisson(5, (500, 200)))
table = biom.Table(
    sparse_data,
    observation_ids=[f"OTU_{i}" for i in range(500)],
    sample_ids=[f"Sample_{j}" for j in range(200)],
)
biom.util.hdf5.overwrite("sparse_table.biom", table)

# Method 3: With taxonomy metadata
taxonomy = [
    {"taxonomy": ["k__Bacteria", "p__Firmicutes", "c__Clostridia",
                   "o__Lachnospiraceae", "g__Roseburia", f"s__sp_{i}"]}
    for i in range(100)
]
table = biom.Table(
    df.values.T,
    observation_ids=df.index.tolist(),
    sample_ids=df.columns.tolist(),
    observation_metadata=taxonomy,
)
biom.util.hdf5.overwrite("table_with_taxonomy.biom", table)
```

### 3. Filtering and Subsetting

```python
import biom
import numpy as np

table = biom.load_table("feature_table.biom")

# Filter by prevalence (keep features present in at least N samples)
def filter_by_prevalence(table, min_samples=5):
    prevalence = (table.matrix_data > 0).sum(axis=1).A1 if hasattr(table.matrix_data, "A1") \
                 else (table.matrix_data > 0).sum(axis=1)
    keep = np.where(prevalence >= min_samples)[0]
    return table.filter(keep, axis="observation")

filtered = filter_by_prevalence(table, min_samples=10)
print(f"Filtered: {len(table.ids(axis='observation'))} -> {len(filtered.ids(axis='observation'))} features")

# Filter by total abundance
def filter_by_abundance(table, min_reads=100):
    abundance = np.array(table.sum(axis="observation")).flatten()
    keep = np.where(abundance >= min_reads)[0]
    return table.filter(keep, axis="observation")

# Subset to specific samples
sample_subset = table.ids(axis="sample")[:10]
subset_table = table.filter(sample_subset, axis="sample")

# Save filtered table
biom.util.hdf5.overwrite("filtered_table.biom", subset_table)
```

## Common Academic Workflow: QIIME 2 to Python Pipeline

```python
import biom
import pandas as pd
from skbio.stats.composition import multipicewise_transform as mpt
from scipy.stats import spearmanr
import numpy as np

# 1. Export from QIIME 2: qiime tools export feature-table.biom.qza --output-dir exported/
# Or convert: biom convert input.biom -o output.tsv --to-tsv

# 2. Load BIOM table
table = biom.load_table("exported/feature-table.biom")
df = table.to_dataframe(dense=True)

# 3. Rarefy for alpha diversity (standard practice)
def rarefy(df, depth=10000):
    """Rarefy all samples to equal sequencing depth."""
    np.random.seed(42)
    rarefied = {}
    for col in df.columns:
        counts = df[col].values
        total = counts.sum()
        if total >= depth:
            idx = np.random.choice(len(counts), size=depth, replace=False, p=counts / total)
            rarefied_counts = np.zeros(len(counts))
            for i in idx:
                rarefied_counts[i] += 1
            rarefied[col] = rarefied_counts
        else:
            rarefied[col] = counts  # keep as-is if below depth
    return pd.DataFrame(rarefied, index=df.index)

rarefied_df = rarefy(df, depth=10000)

# 4. Alpha diversity (Shannon index)
def shannon_diversity(counts):
    """Compute Shannon diversity index."""
    proportions = counts / counts.sum()
    proportions = proportions[proportions > 0]
    return -np.sum(proportions * np.log(proportions))

alpha_div = rarefied_df.apply(shannon_diversity)
print(f"Shannon diversity range: {alpha_div.min():.2f} - {alpha_div.max():.2f}")

# 5. Beta diversity (Bray-Curtis)
from scipy.spatial.distance import pdist, squareform
bc_distances = pdist(rarefied_df.T, metric="braycurtis")
bc_matrix = pd.DataFrame(squareform(bc_distances),
                          index=rarefied_df.columns,
                          columns=rarefied_df.columns)
print(f"Beta diversity matrix: {bc_matrix.shape}")

# 6. Save results
alpha_div.to_csv("alpha_diversity.csv")
bc_matrix.to_csv("beta_diversity.csv")
```

## Best Practices

1. Rarefy samples to equal sequencing depth before computing alpha diversity to control for sampling effort
2. Use sparse matrix representation for large tables (>10K features) to save memory
3. Filter low-prevalence features (<1% of samples) before analysis to reduce noise
4. Store taxonomy metadata in the BIOM file rather than separate TSV files for portability
5. Use HDF5 format (.biom) rather than JSON for large tables (faster I/O, smaller files)

## Common Pitfalls

1. **Zero-inflation**: Microbiome data is extremely sparse; use appropriate statistical methods (zero-inflated models, compositional approaches)
2. **Compositional nature**: Relative abundances sum to 1; use CLR or ILR transforms before correlation or ML
3. **Rarefaction depth**: Choose a depth that retains most samples; discarding low-depth samples biases results
4. **BIOM version mismatch**: QIIME 2 v2023+ uses BIOM 2.1; ensure your biom-format package supports it
